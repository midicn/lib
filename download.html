快速上手

  1. 下载 meta.zip，用 catalog.json 按许可 / 时期 / 时长挑出你要的曲目。
  2. 下载对应的内容包（例如只要可商用就下 main.zip），解压到同一目录。
  3. 用 f 字段定位文件；du（时长）与 nn（音符数）可直接用于训练集筛选。
import json, shutil
from pathlib import Path

cat = json.load(open('meta/catalog.json', encoding='utf-8'))

# 只要「可商用 + 60~180 秒 + 音符数 > 200」的曲目
picks = [t for t in cat['tracks']
         if t['z'] == 'main' and t['l'] in ('CC0-1.0', 'CC-BY-4.0', 'PD', 'OPEN')
         and 60 <= (t['du'] or 0) <= 180 and (t['nn'] or 0) > 200]

Path('out').mkdir(exist_ok=True)
for t in picks[:500]:
    shutil.copy(Path('midicn-lib-v1.4') / t['f'], 'out')
print(f'挑出 {len(picks):,} 首，已复制 500 首')
使用前请阅读 许可与法律：CC BY 系列必须署名并注明来源；CC BY-NC 系列禁止商用。

Quick Start

  1. Download meta.zip and use catalog.json to select tracks by license / period / duration.
  2. Download the matching content package (e.g. main.zip for commercial-ready), unzip into the same directory.
  3. Locate files via the f field; du (duration) and nn (note count) are ready for dataset filtering.
import json, shutil
from pathlib import Path

cat = json.load(open('meta/catalog.json', encoding='utf-8'))

# commercial-ready && 60-180s && more than 200 notes
picks = [t for t in cat['tracks']
         if t['z'] == 'main' and t['l'] in ('CC0-1.0', 'CC-BY-4.0', 'PD', 'OPEN')
         and 60 <= (t['du'] or 0) <= 180 and (t['nn'] or 0) > 200]

Path('out').mkdir(exist_ok=True)
for t in picks[:500]:
    shutil.copy(Path('midicn-lib-v1.4') / t['f'], 'out')
print(f'{len(picks):,} matched, 500 copied')
